WITH 
 MEMBER [Measures].[Revenue This Month] AS 
   [Measures].[Revenue]
SELECT
  { [Measures].[Revenue This Month] } ON COLUMNS,
  [Dim Time].[Month].[Month].Members ON ROWS
FROM [Foodmart]
WHERE ( [Dim Time].[Year].&[1997])

-- Tăng trưởng doanh thu YoY (Year-over-Year)
WITH
 MEMBER [Measures].[Revenue Current Year]  AS [Measures].[Revenue]
 MEMBER [Measures].[Revenue Prior Year]    AS (
     [Dim Time].[Year].CurrentMember.PrevMember,
     [Measures].[Revenue]
   )
 MEMBER [Measures].[YoY Growth %] AS 
   IIF(
     [Measures].[Revenue Prior Year] = 0,
     NULL,
     ([Measures].[Revenue Current Year] - [Measures].[Revenue Prior Year])
     / [Measures].[Revenue Prior Year]
   )
SELECT
  { 
    [Measures].[Revenue Current Year],
    [Measures].[Revenue Prior Year],
    [Measures].[YoY Growth %]
  } ON COLUMNS,
  [Dim Time].[Year].Members ON ROWS
FROM [Foodmart]
-- Top 10 sản phẩm theo doanh thu theo năm
SELECT
    TOPCOUNT(
        [Dim Product].[Product Name].Members,
        10,
        ([Measures].[Revenue])
    ) ON ROWS,

    {[Measures].[Revenue]} ON COLUMNS

FROM [Foodmart]

WHERE ([Dim Time].[Year].[1997])

-- Tỷ lệ trả hàng (Return Rate) theo sản phẩm
WITH 
MEMBER [Measures].[Return Rate] AS
    ([Measures].[Quantity Return]) / ([Measures].[Quantity]) * 100

SELECT
    [Dim Product].[Product Name].Members ON ROWS,

    {
        [Measures].[Quantity],
        [Measures].[Quantity Return] as [Return Quantity],
        [Measures].[Return Rate]
    } ON COLUMNS

FROM [Foodmart]
